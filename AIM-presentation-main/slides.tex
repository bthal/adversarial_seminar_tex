\attackframe{L-BFGS}

\begin{frame}
  \frametitle{L-BFGS}
  \framesubtitle{Szegedy et al., 2013}

  \taxobox{White-box}{Evasion}{Targeted}

  \begin{block}{}
    \begin{align*}
      \min_{\delta} \; \|\delta\|_2
      \qquad \text{subject to} \qquad
      f(x + \delta) = t
    \end{align*}
  \end{block}

  \begin{itemize}
    \item $f$ has no gradient --- it returns a discrete class label,  $f(x+\delta) = t$ is a hard, non-differentiable condition
    \item Derivative-based optimisers such as L-BFGS need a gradient to move at all
    \item[$\leadsto$] Reformulate
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{L-BFGS}
  \framesubtitle{Szegedy et al., 2013}

  \taxobox{White-box}{Evasion}{Targeted}

  \begin{block}{}
    \begin{align*}
      \min_{\delta} \; c \, \|\delta\|_2 + L_f(x + \delta,\, t)
    \end{align*}
  \end{block}

  \begin{significance}
    \item First method to \textbf{reliably generate} adversarial examples
    \item Computationally expensive --- far too slow to be practical
    \item[$\leadsto$] It \emph{proved a point} rather than providing a usable attack
  \end{significance}
\end{frame}

\begin{frame}
  \frametitle{Fast Gradient Sign Method}
  \framesubtitle{Goodfellow et al., 2014}

  \taxobox{White-box}{Evasion}{Untargeted}
  
  \begin{itemize}
    \item Linearise the loss function:
    \begin{block}{}
    \[
      \arg\max_{\lvert\lvert\delta\rvert\rvert_\infty\leq\epsilon}L_f(x+\delta, y) = 
      \arg\max_{\lvert\lvert\delta\rvert\rvert_\infty\leq\epsilon}L_f(x, y) + \nabla_x L_f(x, y)^T \delta
    \]
    \end{block}
    \item Maximised by $\delta = \epsilon \cdot sign(\nabla_x L_f(x, y))$
    \item[$\leadsto$] Single step in gradient direction
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{High-dimensional linearity}
  \framesubtitle{Goodfellow et al., 2014}

  \begin{columns}[c]
    \column{0.47\textwidth}
    \begin{itemize}[<+->]
      \item Affine case first: perturbation shifts activation by $w^\top\eta$
      \item Best strategy: $\eta$ aligned with the weights
      \item Score change scales \emph{linearly} with number of dimensions w.r.t. input change
      \item[$\leadsto$] Many imperceptible changes accumulate into
            one large shift
    \end{itemize}

    \column{0.50\textwidth}
    \centering
    \begin{tikzpicture}[>=stealth, x=0.72cm, y=0.72cm]
      % decision boundary
      \draw[rwth-blue-100, thick] (-0.15,0.225) -- (4.15,2.375);
      \node[rwth-blue-100, font=\tiny, above left] at (4.15,2.30) {$w^\top x + b = 0$};
      % w is the normal of the boundary -- shown for reference
      \draw[->, rwth-black-50] (0.60,0.60) -- (0.198,1.405);
      \node[rwth-black-50, font=\tiny, left] at (0.21,1.42) {$w$};
      % x sits on the positive side; stepping along -w crosses by the shortest route
      \draw[->, rwth-red-100, thick] (1.77,2.00) -- (2.40,0.75);
      \node[rwth-red-100, font=\tiny, right] at (2.18,1.46) {$\eta = -\epsilon \operatorname{sign}(w)$};
      \fill[rwth-black-100] (1.77,2.00) circle (1.7pt);
      \node[rwth-black-100, font=\tiny, above left] at (1.82,2.06) {$x$};
      \fill[rwth-red-100] (2.40,0.75) circle (1.7pt);
      \node[rwth-red-100, font=\tiny, below] at (2.45,0.68) {$\tilde{x}$};
    \end{tikzpicture}

    \vskip2pt
    \begin{block}{}
      \vskip-10pt
      \begin{align*}
        w^\top \tilde{x} &= w^\top (x + \eta) \\
                         &= w^\top x + w^\top \eta \\[3pt]
        w^\top \eta &= \epsilon \sum_{i=1}^{n} |w_i| = \epsilon \, m \, n
      \end{align*}
      \vskip-10pt
    \end{block}

    {\tiny\refer{$n$ dimensions, mean weight magnitude $m$}}
  \end{columns}
\end{frame}

\attackframe{ZOO}

\begin{frame}
  \frametitle{ZOO: Zeroth order black-box attacks}

  \taxobox{Black-box}{Evasion}{Targ./Untarg.}

  \begin{itemize}
    \item Black-box attack without substitute model
    \item No quality loss due to transfer
    \item Assumptions: Adversary allowed to query, gets confidence scores
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Coordinate-wise pseudo gradient}

  \taxobox{Black-box}{Evasion}{Targ./Untarg.}
  \begin{block}{}
    \begin{align*}
      \hat{g_i}
      := \frac{\partial l(x)}{\partial x_i}
      \approx \frac{l(x + he_i) - l(x - he_i)}{2h}
    \end{align*}
  \end{block}

  \begin{itemize}
    \item \(l\) loss function based on confidence scores
    \item Two queries per coordinate, full gradient infeasible
    \item -> Coordinate descent using first and/or second-order method
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Proposed loss function}

  \taxobox{Black-box}{Evasion}{Targ./Untarg.}

  \begin{itemize}
    \item Reminder: Carlini-Wagner loss:
      \(\;\ell(x') = \max\!\big(\max_{i\neq t} z_i(x') - z_t(x'),\; 0\big)\) 
    \item Targeted ZOO: \(l(x,t) = \max\{ \max_{i\neq t} \{\log[f(x)]_i\} - \log[f(x)]_t,-\kappa \}\)
    \item Untargeted ZOO: \(l(x) = \max\{ \log[f(x)]_{y} - \max_{i\neq y} \{\log[f(x)]_i\} ,-\kappa \}\)
  \end{itemize}
  \begin{itemize}
    \item \(log\) to estimate logits
  \end{itemize}
\end{frame}

\section{Poisoning attacks}

\attackframe{Hidden Trigger Backdoor Attack}

\begin{frame}
  \frametitle{Hidden Trigger Backdoor Attack}

  \taxobox{Black-box}{???}{}

  \begin{itemize}
    \item Goal: get ML model to misclassify any sample with trigger pattern into target class \(t\)
    \item Similar threat model to Poison Frogs
    \item Restriction: adversary cannot choose labels
    \item -> "Clean-label attack"
  \end{itemize}
  
\end{frame}

\begin{frame}
  \frametitle{Hiding the trigger}

  \taxobox{Black-box}{Evasion}{Targ./Untarg.}

  \begin{itemize}
    \item Sample instances \(s\) \(b\) of target class \(t\)
    \item Apply trigger to source: patched source \(\tilde{s}\) 
    \item Poison instances:
  \end{itemize}
  \begin{block}{}
    \begin{align*}
      \arg\min_x{\lvert \lvert g(x) - g(\tilde{s}) \rvert \rvert_{2}^2} \quad\quad \textbf{s.t.} \quad \lvert \lvert x - b \rvert \rvert_{\infty} < \epsilon
    \end{align*}
  \end{block}

  \begin{itemize}
    \item \(g\): feature representation (second-to-last layer output)
    \item Keep \(x\) close to \(\tilde{s}\) in feature space, but close to \(b\) in pixel space
  \end{itemize}
  
\end{frame}

\subsection{Adversarial attacks as a subject of study}

\begin{frame}[c]
  \frametitle{Adversarial attacks as a subject of study}

  % Explicit overlay specs, not [<+->]: with [<+->] the nested sub-bullets each
  % consume their own step. Sub-bullets share their parent's spec.
  \begin{itemize}
    \item<1-> Not really significant security threat
    \item<2-> Mainly \emph{epistemic significance}:
    \begin{itemize}
      \item<2-> Not: What can an attacker do?
      \item<2-> But: What do they reveal about the targetted system?
    \end{itemize}
  \end{itemize}

  \vfill
  \begin{columns}[t]
    \column{0.47\textwidth}
    \begin{block}<3->{Before}
      \begin{itemize}
        \item Accuracy as near-single metric
        \item Robustness assumed implicitly
      \end{itemize}
    \end{block}

    \column{0.47\textwidth}
    \begin{block}<4->{Afterwards}
      \begin{itemize}
        \item High accuracy $\neq$ human-like understanding
        \item Robustness as property of interest
      \end{itemize}
    \end{block}
  \end{columns}
\end{frame}

\begin{frame}[c]
  \frametitle{Key takeaways}

  \begin{itemize}
    \item Imperceptible perturbation $\rightarrow$ wrong prediction
    \item Transferable across models, architectures, training data
    \item Categories: white/black-box $\cdot$ evasion/poisoning $\cdot$ targeted/untargeted
    \item Phenomenon still not fully explained
    \item Marginal security threat so far but epistemic relevance
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{What is an Adversarial Example?}

  \begin{block}{Definition (Adversarial Example)}
    An \emph{adversarial example} is an \textbf{input to a machine learning model} that has been intentionally perturbed, typically \textbf{in a subtle and often imperceptible manner}, such that the model \textbf{produces an incorrect prediction} while the semantic content of the input remains unchanged.
  \end{block}

  \begin{itemize}
    \item \textbf{Machine learning models}: No quirk of deep networks, but a general property of machine learners
    \item \textbf{Subtle and imperceptible perturbations}: mathematically small under a chosen norm / visually imperceptible to humans
    \item \textbf{Produces an incorrect predition}: The machine learner is fooled where the human would not have been
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Carlini \& Wagner}
  \framesubtitle{Carlini \& Wagner, 2017}

  \taxobox{White-box}{Evasion}{Targeted}

  \begin{block}{}
    \begin{align*}
      \underbrace{\min_{\delta}\; c\,\|\delta\|_2 + L_f(x+\delta,\, t)}_{\text{L-BFGS}}
      \qquad\longrightarrow\qquad
      \underbrace{\min_{\delta}\; \|\delta\|_p + c\cdot \ell(x+\delta)}_{\text{C\&W}}
    \end{align*}
  \end{block}

  \begin{itemize}
    \item $\;\ell(x') = \max\!\big(\max_{i\neq t} z_i(x') - z_t(x'),\; 0\big)$
    \item Built so that $\ell(x') \le 0 \iff$ input is classified as $t$ $\rightarrow$ faithfully encodes the attack goal
    \item $c$ decided through \textbf{binary search}
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Carlini \& Wagner}
  \framesubtitle{Carlini \& Wagner, 2017}

  \taxobox{White-box}{Evasion}{Targeted}

  \begin{block}{}
    \begin{align*}
      x' = \tfrac{1}{2}\big(\tanh(w) + 1\big),
      \qquad w \in \mathbb{R}^m
    \end{align*}
  \end{block}

  \begin{itemize}
    \item Mapping any $w \in \mathbb{R}^m$ into $(0,1)^m$ $\rightarrow$ every $w$
          decodes to a \textbf{valid image} $\rightarrow$ dissolving box-constraint
    \item Optimise $w$ \textbf{unconstrained} with Adam optimiser
  \end{itemize}

  \begin{significance}
    \item Strongest attack of its time across $\ell_0 / \ell_2 / \ell_\infty$
    \item Broke \emph{defensive distillation}, exposing it as \emph{gradient masking}
  \end{significance}
\end{frame}

\begin{frame}
  \frametitle{Poison Frogs}

  \taxobox{Black-box}{Evasion}{Targ./Untarg.}

  \begin{itemize}
    \item Goal: get ML model to misclassify one specific sample \(v\) into target class \(t\)
    \item Assumption: model based on pretrained weights known to the attacker, access to data distribution
    \item Restriction: adversary cannot choose labels
    \item -> "Clean-label attack"
  \end{itemize}
  
\end{frame}

\begin{frame}
  \frametitle{Poison Frogs: poison instances}

  \taxobox{Black-box}{Evasion}{Targ./Untarg.}

  \begin{itemize}
    \item Sample instance \(b\) of target class \(t\)
    \item Poison instance(s):
  \end{itemize}
  \begin{block}{}
    \begin{align*}
      p = \arg\min_x {\lvert \lvert g(x) - g(v) \rvert \rvert_{2}^2 + \lambda \lvert \lvert x - b \rvert \rvert_{2}^2} 
    \end{align*}
  \end{block}

  \begin{itemize}
    \item \(g\): feature representation (second-to-last layer output)
    \item Keep \(x\) close to \(v\) in feature space, but close to \(b\) in pixel space
  \end{itemize}
  
\end{frame}
