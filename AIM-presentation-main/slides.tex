\attackframe{L-BFGS}

\begin{frame}
  \frametitle{L-BFGS}
  \framesubtitle{\cite{szegedy2013intriguing}}

  \taxobox{White-box}{Evasion}{Targeted}

  \begin{block}{}
    \begin{align*}
      \min_{\delta} \; \|\delta\|_2
      \qquad \text{subject to} \qquad
      f(x + \delta) = t
    \end{align*}
  \end{block}

  \begin{itemize}
    \item $f$ has no gradient --- it returns a discrete class label,  $f(x+\delta) = t$ is a hard, non-differentiable condition
    \item Derivative-based optimisers such as L-BFGS need a gradient to move at all
    \item[$\leadsto$] Reformulate
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{L-BFGS}
  \framesubtitle{\cite{szegedy2013intriguing}}

  \taxobox{White-box}{Evasion}{Targeted}

  \begin{block}{}
    \begin{align*}
      \min_{\delta} \; c \, \|\delta\|_2 + L_f(x + \delta,\, t)
    \end{align*}
  \end{block}

  \begin{significance}
    \item First method to \textbf{reliably generate} adversarial examples
    \item Computationally expensive --- far too slow to be practical
    \item[$\leadsto$] It \emph{proved a point} rather than providing a usable attack
  \end{significance}
\end{frame}

\attackframe{Fast Gradient Sign Method (FGSM)}

\begin{frame}
  \frametitle{Fast Gradient Sign Method}
  \framesubtitle{\cite{goodfellow2014explaining}}

  \taxobox{White-box}{Evasion}{Untargeted}
  \begin{block}{}
    
  \end{block}
  \begin{itemize}[<+->]
    \item Previously: Adversarial examples assumed to be caused by 'complex' behaviour
    \item Goodfellow et al.: Linear models have adversarial examples as well
    \item[$\leadsto$] Models may be too linear
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{High-dimensional linearity}
  \framesubtitle{\cite{goodfellow2014explaining}}

  \begin{itemize}[<+->]
      \item Linear model \(w^\top {x}\) perturbed by \(\delta\), \(\lVert \delta \rVert_\infty \leq \epsilon\):
      \(w^\top {x+\delta} = w^\top {x} + w^\top \delta\)
      \item Perturbation shifts activation by \(w^\top \delta\)
      \item Best strategy: $\delta$ aligned with the weights, then:
      \(w^\top \delta = \epsilon \sum_{i=1}^{n} |w_i| = \epsilon \, m \, n\)
      \item Score change scales \emph{linearly} with number of dimensions
      \item[$\leadsto$] Many imperceptible changes accumulate into
            one large shift
    \end{itemize}

    
\end{frame}

\begin{frame}
  \frametitle{Fast Gradient Sign Method}
  \framesubtitle{\cite{goodfellow2014explaining}}

  \taxobox{White-box}{Evasion}{Untargeted}
  
  \begin{itemize}[<+->]
    \item Linearise the loss function:
    \begin{block}{}
    \[
      \arg\max_{\lvert\lvert\delta\rvert\rvert_\infty\leq\epsilon}L_f(x+\delta, y) = 
      \arg\max_{\lvert\lvert\delta\rvert\rvert_\infty\leq\epsilon}L_f(x, y) + \nabla_x L_f(x, y)^T \delta
    \]
    \end{block}
    \item Maximised by $\delta = \epsilon \cdot sign(\nabla_x L_f(x, y))$
    \item[$\leadsto$] Single step in gradient direction
  \end{itemize}
\end{frame}

\attackframe{Projected Gradient Descent (PGD)}

\begin{frame}
  \frametitle{PGD}
  \framesubtitle{\cite{madry2017towards}}

  \taxobox{White-box}{Evasion}{Untargeted}
  \begin{block}{}
    General formulation of adversarial training. Adversary solves
    \[
    \max_{\delta\in S}L_f(x+\delta, y)
    \]
  \end{block}
  \begin{itemize}[<+->]
    \item FGSM: Use \(S=\{\delta \mid \lVert \delta \rVert_\infty \leq \epsilon\}\)
    \item Use projected gradient descent to optimise
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{PGD - a general first-order attack?}
  \framesubtitle{\cite{madry2017towards}}

  \taxobox{White-box}{Evasion}{Untargeted}
  \begin{itemize}[<+->]
    \item Random runs show different maxima, but similar loss values
    \item[$\leadsto$] If better solution exists: difficult to find using first-order methods
    \item Especially relevant for adversarial defenses
  \end{itemize}
  
\end{frame}

\attackframe{Transfer attacks}

\begin{frame}
  \frametitle{Transfer attacks}

  \begin{itemize}[<+->]
    \item Black-box attack pattern
    \item Adversary creates and trains a substitute model for the target
    \item Adversarial examples for substitute model are found (white-box!)
    \item These can be used to attack the target
  \end{itemize}
\end{frame}


\begin{frame}
  \frametitle{Jacobian-Based Dataset Augmentation}
  \framesubtitle{\cite{papernot2017practical}}

  \taxobox{Black-box}{Evasion}{-}

  \begin{itemize}[<+->]
    \item Assumption: No knowledge of model parameters, dataset 
    \linebreak Access to somewhat representative inputs is granted (e.g. natural images) 
    \item How do we construct a suitable substitute? (similar decision boundaries)
    \item Construct an artificial dataset!
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Jacobian-Based Dataset Augmentation}
  \framesubtitle{\cite{papernot2017practical}}

  \taxobox{Black-box}{Evasion}{-}

  \begin{itemize}[<+->]
    \item Use target model as an oracle to label samples 
    \item Strategy for sample selection is necessary
    \item Idea: iteratively add meaningful samples to dataset
    \item Use Jacobian for a heuristic
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Heuristic for meaningful directions}
  \framesubtitle{\cite{papernot2017practical}}

  \taxobox{Black-box}{Evasion}{-}

  \begin{block}{}
    \begin{align*}
      x = x + \alpha \cdot sign(J_f(x)[o(x)])
    \end{align*}
  \end{block}


  \begin{itemize}
    \item \(J_f(x)\): total derivative
    \item \(J_f(x)[o(x)]\): total derivative, column of the correct label
    \item[$\leadsto$] We move into the class region (according to the substitute)!
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Jacobian-Based Dataset Augmentation}
  \framesubtitle{\cite{papernot2017practical}}

  \taxobox{Black-box}{Evasion}{-}

  \begin{itemize}[<+->]
    \item 1) Collect samples somewhat related to the input domain
    \item 2) Choose architecture for substitute
    \item 3) Target model labels current samples
    \item 4) Train substitute
    \item 5) Expand dataset: \( S_{N+1} := \{x = x + \alpha \cdot sign(J_f(x)[o(x)]) \mid x \in S_N\} \cup S_N\)
    \item Repeat (3) onwards
  \end{itemize}
\end{frame}



\attackframe{ZOO}

\begin{frame}
  \frametitle{ZOO: Zeroth order black-box attacks}
  \framesubtitle{\cite{chen2017zoo}}
  \taxobox{Black-box}{Evasion}{Targ./Untarg.}

  \begin{itemize}[<+->]
    \item Black-box attack without substitute model
    \item[$\leadsto$] No quality loss due to transfer
    \item Assumptions: Adversary allowed to query, gets confidence scores
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{ZOO: Coordinate-wise pseudo gradient}
  \framesubtitle{\cite{chen2017zoo}}
  \taxobox{Black-box}{Evasion}{Targ./Untarg.}
  \begin{block}{}
    \begin{align*}
      \hat{g_i}
      := \frac{\partial l(x)}{\partial x_i}
      \approx \frac{l(x + he_i) - l(x - he_i)}{2h}
    \end{align*}
  \end{block}

  \begin{itemize}[<+->]
    \item \(l\) loss function based on confidence scores
    \item Two queries per coordinate, full gradient infeasible
    \item[$\leadsto$] Coordinate descent% using first and/or second-order method
  \end{itemize}
\end{frame}

\attackframe{Hidden Trigger Backdoor Attack}

\begin{frame}

  \framesubtitle{\cite{saha2020hidden}}

    \item Restriction: adversary cannot choose labels ("Clean-label attack")

\begin{frame}
  \frametitle{Hiding the trigger}
  \framesubtitle{\cite{saha2020hidden}}
  \taxobox{White-box}{Poisoning}{Targeted}

  \begin{itemize}
    \item Sample instances \(s\) \(b\) of target class \(t\)
    \item Apply trigger to source: patched source \(\tilde{s}\) 
    \item Poison instances:
  \end{itemize}
  \begin{block}{}
    \begin{align*}
      \arg\min_x{\lvert \lvert g(x) - g(\tilde{s}) \rvert \rvert_{2}^2} \quad\quad \textbf{s.t.} \quad \lvert \lvert x - b \rvert \rvert_{\infty} < \epsilon
    \end{align*}
  \end{block}

  \begin{itemize}[<+->]
    \item \(g\): feature representation (second-to-last layer output)
    \item Keep \(x\) close to \(\tilde{s}\) in feature space, but close to \(b\) in pixel space
  \end{itemize}
  
\end{frame}

\attackframe{Narcissus}

\begin{frame}
  \frametitle{Narcissus}
  \framesubtitle{\cite{zeng2023narcissus}}
  \taxobox{Black-box}{Poisoning}{Targeted}

  \begin{itemize}[<+->]
    \item Goal: get ML model to misclassify any sample with trigger pattern into target class \(t\)
    \item Assumption: no access to model or data distribution
    \item But: Access to "public out of distribution" data \(D_{POOD}\), target class examples \(D_t\)
    \item Restriction: adversary cannot choose labels ("clean-label attack")
    \item Now: Optimise the trigger pattern!
  \end{itemize}
  
\end{frame}

\begin{frame}
  \frametitle{Narcissus}
  \framesubtitle{\cite{zeng2023narcissus}}
  \taxobox{Black-box}{Poisoning}{Targeted}

  \begin{itemize}[<+->]
    \item Assume oracle \(f_{orc}\), which is the victim's model without poisoning
    \item Choose trigger to maximise oracle's confidence on target class: \(\delta^{*} = \arg\min_{\delta\in S}{\sum_{(x,t)\in D_t}{L_{f_{orc}}(x+\delta, t)}}\)
    \item Trigger now is related to the target class!
    \item However: No access to \(f_{orc}\)
  \end{itemize}
  
\end{frame}

\begin{frame}
  \frametitle{Narcissus}
  \framesubtitle{\cite{zeng2023narcissus}}
  \taxobox{Black-box}{Poisoning}{Targeted}

  \begin{itemize}[<+->]
    \item Train surrogate \(f_{sur}\) on \(D_{POOD}\), \(D_t\)
    \item Use PGD to find \(\delta^{*} = \arg\min_{\delta\in S}{\sum_{(x,t)\in D_t}{L_{f_{sur}}(x+\delta, t)}}\)
    \item Add trigger to portion of \(D_t\), give to victim
    \item Exploit: Add trigger to desired input
  \end{itemize}
  
\end{frame}

\begin{frame}[c]

\begin{frame}[c]
  \frametitle{Predictive and robust features}
  \framesubtitle{\cite{ilyas2019adversarial}}

  \begin{columns}[t]
    \column{0.47\textwidth}
    \centering
    \textbf{Predictive}\\[3pt]
    \parbox[t][2.6em]{\linewidth}{\centering\small
      Correlated with the label in expectation.}\\[6pt]

    \column{0.47\textwidth}
    \centering
    \uncover<2->{\textbf{Robust}}\\[3pt]
    \parbox[t][2.6em]{\linewidth}{\centering\small
      \uncover<2->{Stays correlated \emph{under} adversarial perturbation.}}\\[6pt]
  \end{columns}

  \vfill
  \begin{itemize}[<3->]
    \item[$\leadsto$] \textbf{Predictive but non-robust:} Standard training happily relies on such features
  \end{itemize}
\end{frame}

\begin{frame}[c]
  \frametitle{Experiment: "Mislabeled" training dataset}
  \framesubtitle{\cite{ilyas2019adversarial}}

\begin{frame}
  \frametitle{What is an Adversarial Example?}

  \begin{block}{Definition (Adversarial Example)}
    An \emph{adversarial example} is an \textbf{input to a machine learning model} that has been intentionally perturbed, typically \textbf{in a subtle and often imperceptible manner}, such that the model \textbf{produces an incorrect prediction} while the semantic content of the input remains unchanged.
  \end{block}

  \begin{itemize}
    \item \textbf{Machine learning models}: No quirk of deep networks, but a general property of machine learners
    \item \textbf{Subtle and imperceptible perturbations}: mathematically small under a chosen norm / visually imperceptible to humans
    \item \textbf{Produces an incorrect predition}: The machine learner is fooled where the human would not have been
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Carlini \& Wagner}
  \framesubtitle{\cite{carlini2017towards}}

  \taxobox{White-box}{Evasion}{Targeted}

  \begin{block}{}
    \begin{align*}
      \underbrace{\min_{\delta}\; c\,\|\delta\|_2 + L_f(x+\delta,\, t)}_{\text{L-BFGS}}
      \qquad\longrightarrow\qquad
      \underbrace{\min_{\delta}\; \|\delta\|_p + c\cdot \ell(x+\delta)}_{\text{C\&W}}
    \end{align*}
  \end{block}

  \begin{itemize}
    \item $\;\ell(x') = \max\!\big(\max_{i\neq t} z_i(x') - z_t(x'),\; 0\big)$
    \item Built so that $\ell(x') \le 0 \iff$ input is classified as $t$ $\rightarrow$ faithfully encodes the attack goal
    \item $c$ decided through \textbf{binary search}
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Carlini \& Wagner}
  \framesubtitle{\cite{carlini2017towards}}

  \taxobox{White-box}{Evasion}{Targeted}

  \begin{block}{}
    \begin{align*}
      x' = \tfrac{1}{2}\big(\tanh(w) + 1\big),
      \qquad w \in \mathbb{R}^m
    \end{align*}
  \end{block}

  \begin{itemize}
    \item Mapping any $w \in \mathbb{R}^m$ into $(0,1)^m$ $\rightarrow$ every $w$
          decodes to a \textbf{valid image} $\rightarrow$ dissolving box-constraint
    \item Optimise $w$ \textbf{unconstrained} with Adam optimiser
  \end{itemize}

  \begin{significance}
    \item Strongest attack of its time across $\ell_0 / \ell_2 / \ell_\infty$
    \item Broke \emph{defensive distillation}, exposing it as \emph{gradient masking}
  \end{significance}
\end{frame}

\begin{frame}
  \frametitle{Poison Frogs}

  \taxobox{Black-box}{Evasion}{Targ./Untarg.}

  \begin{itemize}
    \item Goal: get ML model to misclassify one specific sample \(v\) into target class \(t\)
    \item Assumption: model based on pretrained weights known to the attacker, access to data distribution
    \item Restriction: adversary cannot choose labels
    \item -> "Clean-label attack"
  \end{itemize}
  
\end{frame}

\begin{frame}
  \frametitle{Poison Frogs: poison instances}

  \taxobox{Black-box}{Evasion}{Targ./Untarg.}

  \begin{itemize}
    \item Sample instance \(b\) of target class \(t\)
    \item Poison instance(s):
  \end{itemize}
  \begin{block}{}
    \begin{align*}
      p = \arg\min_x {\lvert \lvert g(x) - g(v) \rvert \rvert_{2}^2 + \lambda \lvert \lvert x - b \rvert \rvert_{2}^2} 
    \end{align*}
  \end{block}

  \begin{itemize}
    \item \(g\): feature representation (second-to-last layer output)
    \item Keep \(x\) close to \(v\) in feature space, but close to \(b\) in pixel space
  \end{itemize}
  
\end{frame}
