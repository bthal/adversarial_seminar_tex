\section{What is an adversarial example?}
% ==========================================================================

\begin{frame}
  \frametitle{Definition}

  \begin{block}{Definition (Adversarial Example)}
    An \emph{adversarial example} is an \textbf{input to a machine learning model}
    that has been \textbf{intentionally perturbed}, typically \textbf{in a subtle
    and often imperceptible manner}, such that the model produces an incorrect
    prediction while the semantic content of the input remains unchanged.
  \end{block}

  \pause
  \begin{itemize}
    \item \textbf{Machine learning models} --- no quirk of deep networks, but a general
          property of machine learners.
    \item \textbf{Intentional perturbations} --- generated systematically through
          optimisation and exploited by an adversary, not stumbled upon.
    \item \textbf{Subtle and imperceptible perturbations} --- mathematically small under
          a chosen norm, and visually indistinguishable to humans; the latter is
          asserted far more often than it is rigorously validated.
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{L-BFGS}

  \taxobox{White-box}{Evasion}{Targeted}{$\ell_2$}

  \begin{block}{}
    \begin{align*}
      \min_{\delta} \; \|\delta\|_2
      \qquad \text{subject to} \qquad
      f(x + \delta) = t
    \end{align*}
  \end{block}

  \begin{itemize}
    \item $f$ has no gradient --- it returns a discrete class label,  $f(x+\delta) = t$ is a hard, non-differentiable condition.
    \item Derivative-based optimisers such as L-BFGS need a gradient to move at all.
    \item[$\leadsto$] Reformulate.
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{L-BFGS}

  \taxobox{White-box}{Evasion}{Targeted}{$\ell_2$}

  \begin{block}{}
    \begin{align*}
      \min_{\delta} \; c \, \|\delta\|_2 + L_f(x + \delta,\, t)
    \end{align*}
  \end{block}

  \begin{significance}
    \item First method to \textbf{reliably generate} adversarial examples.
    \item Computationally expensive --- far too slow to be practical.
    \item[$\leadsto$] It \emph{proved a point} rather than providing a usable attack.
  \end{significance}
\end{frame}

\begin{frame}
  \frametitle{Fast Gradient Sign Method}

  \taxobox{White-box}{Evasion}{Untargeted}{$\ell_\infty$}
  
  \begin{itemize}
    \item Linearise the loss function:
    \begin{block}{}
    \[
      \arg\max_{\lvert\lvert\delta\rvert\rvert_\infty\leq\epsilon}L_f(x+\delta, y) = 
      \arg\max_{\lvert\lvert\delta\rvert\rvert_\infty\leq\epsilon}L_f(x, y) + \nabla_x L_f(x, y)^T \delta
    \]
    \end{block}
    \item Maximised by $\delta = \epsilon \cdot sign(\nabla_x L_f(x, y))$
    \item[$\leadsto$] single step in gradient direction
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{DeepFool}
  \framesubtitle{The affine, binary case}

  \taxobox{White-box}{Evasion}{Untargeted}{$\ell_2$}

  \begin{columns}[c]
    \column{0.55\textwidth}
    \begin{itemize}
      \item For an affine $f(x) = w^\top x + b$, the decision boundary
            $\mathcal{F}$ is a \textbf{hyperplane}.
      \item Minimal perturbation is the \textbf{orthogonal projection} of $x_0$
            onto $\mathcal{F}$ --- closed form, no search.
    \end{itemize}

    \vskip-2pt
    \begin{block}{}
      \vskip-6pt
      \begin{align*}
        r_*(x_0) = -\,\frac{f(x_0)}{\|w\|_2^{2}}\, w
      \end{align*}
      \vskip-6pt
    \end{block}

\begin{frame}
  \frametitle{DeepFool}
  \framesubtitle{The general, multiclass case}

  \taxobox{White-box}{Evasion}{Untargeted}{$\ell_2$}

  \begin{columns}[c]
    \column{0.55\textwidth}
    \begin{itemize}
      \item \textbf{Non-linear:} iterative $\rightarrow$ linearise the
            score function at current point (first-order Taylor), project onto that
            hyperplane, repeat until label flips.
      \item \textbf{Multiclass:}
            Linearise \emph{every} class boundary, project onto the \textbf{closest}.
    \end{itemize}

    \column{0.43\textwidth}
    \centering
    \includegraphics[width=\textwidth, height=0.5\textheight, keepaspectratio]{nonlinear-multiclass.png}
  \end{columns}

  \vskip2pt
  \begin{significance}
    \item Gave problem a \textbf{geometric reading}: robustness as distance
          to the decision boundary.
    \item First to \textbf{extend to any $\ell_p$ norm}, $p \in [1, \infty)$.
  \end{significance}
\end{frame}

\begin{frame}
  \frametitle{Carlini \& Wagner}

  \taxobox{White-box}{Evasion}{Targeted}{$\ell_0 / \ell_2 / \ell_\infty$}

  \begin{block}{}
    \begin{align*}
      \underbrace{\min_{\delta}\; c\,\|\delta\|_2 + L_f(x+\delta,\, t)}_{\text{L-BFGS}}
      \qquad\longrightarrow\qquad
      \underbrace{\min_{\delta}\; \|\delta\|_p + c\cdot \ell(x+\delta)}_{\text{C\&W}}
    \end{align*}
  \end{block}

  \begin{itemize}
    \item $\;\ell(x') = \max\!\big(\max_{i\neq t} z_i(x') - z_t(x'),\; 0\big)$.
    \item Built so that $\ell(x') \le 0 \iff$ input is classified as $t$ $\rightarrow$ faithfully encodes the attack goal.
    \item $c$ decided through \textbf{binary search}
  \end{itemize}
\end{frame}

\begin{frame}

  \taxobox{White-box}{Evasion}{Targeted}{$\ell_0 / \ell_2 / \ell_\infty$}

      x' = \tfrac{1}{2}\big(\tanh(w) + 1\big),

  \end{significance}

\subsection{Real-life adversarial attacks}

\begin{frame}[c]
  \frametitle{Real-life adversarial attacks}

  \begin{itemize}
    \item Confirmed malicious attacks on real institutions or individuals --- strikingly rare
    \item Two candidate explanations:
    \begin{enumerate}
      \item they slip through unnoticed, logged as ordinary misclassifications
      \item they are simply of little use to actual attackers
    \end{enumerate}
    \item Case for the second:
    \begin{itemize}
      \item crafting them reliably needs ML \emph{and} security expertise
      \item DDoS, character obfuscation, packing a binary --- cheap, fast, effective
      \item[$\leadsto$] picking a lock where a crowbar is cheaper and works just as well
    \end{itemize}
  \end{itemize}
\end{frame}

      \node[rwth-black-50, font=\tiny, left] at (0.21,1.42) {$w$};
      % x sits on the positive side; stepping along -w crosses by the shortest route
      \draw[->, rwth-red-100, thick] (1.77,2.00) -- (2.40,0.75);
      \node[rwth-red-100, font=\tiny, right] at (2.18,1.46) {$\eta = -\epsilon \operatorname{sign}(w)$};
      \fill[rwth-black-100] (1.77,2.00) circle (1.7pt);
      \node[rwth-black-100, font=\tiny, above left] at (1.82,2.06) {$x$};
      \fill[rwth-red-100] (2.40,0.75) circle (1.7pt);
      \node[rwth-red-100, font=\tiny, below] at (2.45,0.68) {$\tilde{x}$};
    \end{tikzpicture}

    \vskip2pt
    \begin{block}{}
      \vskip-10pt
      \begin{align*}
        w^\top \tilde{x} &= w^\top (x + \eta) \\
                         &= w^\top x + w^\top \eta \\[3pt]
        w^\top \eta &= \epsilon \sum_{i=1}^{n} |w_i| = \epsilon \, m \, n
      \end{align*}
      \vskip-10pt
    \end{block}

    {\tiny\refer{$n$ dimensions, mean weight magnitude $m$}}
  \end{columns}
\end{frame}

  \begin{itemize}
    \item Not really significant security threat
    \item Mainly \emph{epistemic significance}: not about what attacker can do but more
          about what they reveal about the systems they target
    \item Before them: accuracy was almost only metric, robustness assumed implicitly
    \item After them:
    \begin{itemize}
      \item high test accuracy $\neq$ grasping the task as a human would
      \item a more intricate account of how learners operate
      \item robustness as a distinct property of interest
    \end{itemize}
